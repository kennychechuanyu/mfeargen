model {
  for (i in 1:Nparticipants) {
    gp[i] ~ dcat(pi[1:4])
    tau_obs[i] <- ifelse(gp[i] == 1, pow(sigma_obs[2], -2), pow(sigma_obs[1], -2))

    alpha_stable_raw[i] ~ dnorm(mu_alpha, 1 / tau_alpha_sq) T(0, 1)
    alpha_stable[i] <- ifelse(gp[i] == 1, 0, alpha_stable_raw[i])
    lambda_stable_raw[i] ~ dnorm(mu_lambda, 1 / tau_lambda_sq) T(0, )
    lambda_stable[i] <- ifelse(gp[i] == 1, 0,
      ifelse(gp[i] == 2, min(lambda_stable_raw[i], lambda_threshold),
             max(lambda_stable_raw[i], lambda_threshold)))
    w0_stable[i] ~ dnorm(mu_w0, 1 / tau_w0_sq)
    w1_stable_raw[i] ~ dnorm(mu_w1, 1 / tau_w1_sq) T(0, )
    w1_stable[i] <- ifelse(gp[i] == 1, 0, w1_stable_raw[i])

    for (t in 1:Nsessions[i]) {
      alpha_raw[i,t] ~ dnorm(alpha_stable[i], 1 / sigma_alpha_sq) T(0, 1)
      alpha[i,t] <- ifelse(gp[i] == 1, 0, alpha_raw[i,t])
      lambda_raw[i,t] ~ dnorm(lambda_stable[i], 1 / sigma_lambda_sq) T(0, )
      lambda[i,t] <- ifelse(gp[i] == 1, 0,
        ifelse(gp[i] == 2, min(lambda_raw[i,t], lambda_threshold),
               max(lambda_raw[i,t], lambda_threshold)))
      w0[i,t] ~ dnorm(w0_stable[i], 1 / sigma_w0_sq)
      w1_raw[i,t] ~ dnorm(w1_stable[i], 1 / sigma_w1_sq) T(0, )
      w1[i,t] <- ifelse(gp[i] == 1, 0, w1_raw[i,t])

      # Each session has its own conditioning history.
      v[i,t,1] <- 0
      for (j in 1:Ntrials[i,t]) {
        y[i,t,j] ~ dnorm(theta[i,t,j], tau_obs[i])
        loglik[i,t,j] <- logdensity.norm(y[i,t,j], theta[i,t,j], tau_obs[i])
        y_pre[i,t,j] ~ dnorm(theta[i,t,j], tau_obs[i])
        theta[i,t,j] <- A + (K - A) * ilogit(w0[i,t] + w1[i,t] * g[i,t,j])
        distance[i,t,j] <- ifelse(gp[i] == 3, d_phy[i,t,j], d_per[i,t,j])
        g[i,t,j] <- v[i,t,j] * exp(-lambda[i,t] * distance[i,t,j])
        v[i,t,j+1] <- v[i,t,j] + k[i,t,j] * alpha[i,t] * (r[i,t,j] - v[i,t,j])
      }
    }
  }

  mu_alpha ~ dbeta(alpha_a_pop, alpha_b_pop) T(0.01, 0.99)
  mu_lambda ~ dlnorm(lambda_mu_pop, 1 / pow(lambda_sigma_pop, 2))
  mu_w0 ~ dnorm(w0_mu_pop, 1 / pow(w0_sigma_pop, 2))
  mu_w1 ~ dgamma(w1_a_pop, w1_b_pop) T(0.1, )

  # These are variances on the underlying normal parameter scales.
  tau_alpha_sq ~ dgamma(2, 1)
  tau_lambda_sq ~ dgamma(2, 1)
  tau_w0_sq ~ dgamma(2, 1)
  tau_w1_sq ~ dgamma(2, 1)
  sigma_alpha_sq ~ dgamma(2, 1)
  sigma_lambda_sq ~ dgamma(2, 1)
  sigma_w0_sq ~ dgamma(2, 1)
  sigma_w1_sq ~ dgamma(2, 1)

  sigma_obs[1] ~ dunif(0.1, 2)
  sigma_obs[2] ~ dunif(1.5, 4)
  pi[1:4] ~ ddirch(pi_alpha[1:4])

  alpha_a_pop <- alpha_mu_hyper * alpha_kappa_hyper
  alpha_b_pop <- (1 - alpha_mu_hyper) * alpha_kappa_hyper
  alpha_mu_hyper ~ dbeta(1, 1)
  alpha_kappa_hyper ~ dunif(1, 50)
  lambda_mu_pop ~ dnorm(0, 4)
  lambda_sigma_pop ~ dunif(0.01, 1)
  w0_mu_pop ~ dnorm(0, 0.01)
  w0_sigma_pop ~ dgamma(2, 1)
  w1_a_pop ~ dgamma(2, 1)
  w1_b_pop ~ dgamma(2, 1)

  stability_alpha <- tau_alpha_sq / (tau_alpha_sq + sigma_alpha_sq)
  stability_lambda <- tau_lambda_sq / (tau_lambda_sq + sigma_lambda_sq)
  stability_w0 <- tau_w0_sq / (tau_w0_sq + sigma_w0_sq)
  stability_w1 <- tau_w1_sq / (tau_w1_sq + sigma_w1_sq)
}
