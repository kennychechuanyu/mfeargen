model {
  for (i in 1:Nparticipants) {
    for (j in 1:Ntrials) {
      y[i,j] ~ dnorm(theta[i,j], tau_obs[i])
      theta[i,j] <- rating_scale[1] + (rating_scale[2] - rating_scale[1]) * ilogit(w0[i] + w1[i] * g[i,j])
      loglik[i,j] <- logdensity.norm(y[i,j], theta[i,j], tau_obs[i])
      y_pre[i,j] ~ dnorm(theta[i,j], tau_obs[i])
      g[i,j] <- v_plus[i,j] * s_plus[i,j] + v_minus[i,j] * s_minus[i,j]
      s_plus[i,j] <- exp(-lambda[i] * ifelse(gp[i] == 3, d_p_phy[i,j], d_p_per[i,j]))
      s_minus[i,j] <- exp(-lambda[i] * ifelse(gp[i] == 3, d_m_phy[i,j], d_m_per[i,j]))
      v_plus[i,j+1] <- ifelse(gp[i] == 1, 0, v_plus[i,j] + k_plus[i,j] * alpha[i] * (r_plus[i,j] - v_plus[i,j]))
      v_minus[i,j+1] <- ifelse(gp[i] == 1, 0, v_minus[i,j] + k_minus[i,j] * alpha[i] * (r_minus[i,j] - v_minus[i,j]))
    }
    v_plus[i,1] <- 0
    v_minus[i,1] <- 0

    # Groups: non-learner, overgeneralizer, physical, perceptual.
    gp[i] ~ dcat(pi[1:4])
    tau_obs[i] <- ifelse(gp[i] == 1, pow(sigma[2], -2), pow(sigma[1], -2))
    lambda_1[i] ~ dnorm(lambda_mu, pow(lambda_sigma, -2)) T(lambda_threshold,)
    lambda_2[i] ~ dnorm(lambda_mu, pow(lambda_sigma, -2)) T(0,lambda_threshold)
    lambda[i] <- ifelse(gp[i] == 1, 0, ifelse(gp[i] == 2, lambda_2[i], lambda_1[i]))
    alpha_1[i] ~ dbeta(alpha_a, alpha_b) T(1e-9,1-1e-9)
    alpha[i] <- ifelse(gp[i] == 1, 0, alpha_1[i])
    w0[i] ~ dnorm(w0_mu, pow(w0_sigma, -2))
    w1_1[i] ~ dgamma(w1_a, w1_b) T(1,)
    w1[i] <- ifelse(gp[i] == 1, 0, w1_1[i])
  }

  alpha_a <- alpha_mu * alpha_kappa
  alpha_b <- (1 - alpha_mu) * alpha_kappa
  lambda_mu ~ dlnorm(0, 4)
  lambda_sigma ~ dunif(1e-9, 1)
  alpha_mu ~ dbeta(1, 1) T(1e-9,1-1e-9)
  alpha_kappa ~ dunif(1, 10)
  sigma[1] ~ dunif(1e-9, 1.5)
  sigma[2] ~ dunif(1.5, 3)
  pi[1:4] ~ ddirch(c(1, 1, 1, 1))
  w0_mu ~ dnorm(0, 0.01)
  w0_sigma ~ dgamma(2, 1)
  w1_a ~ dgamma(2, 1)
  w1_b ~ dgamma(2, 1)
}
