data{
	  A <- 0
      K <- 100
}

model{
   for (i in 1: Nparticipants){ # Participant loop start
      for (j in 1: Ntrials){ # Trial loop start


      #------------------------------------------------------------------------------------------------------------#
      y [i,j] ~ dnorm (theta [i,j], 1/sigma [zn [i]]^2)
	  #- response level
	  #-- theta is mean generalized response based on physical distance
	  #-- sigma is the response uncertainty

      theta [i,j] <- A + (K-A) / (1 + exp (- (w0 [i] + w1 [i] * g [i,j])))
	  #--- Non-linear transfromation (latent - observed scale)

      #------------------------------------------------------------------------------------------------------------#

 #-#-#-#-#-# Generalization process #-#-#-#-#-#
      #------------------------------------------------------------------------------------------------------------#
      g [i,j] <- v_plus [i,j] * s_plus [i,j] + v_minus [i,j] * s_minus [i,j]

      s_plus [i,j] <-  ifelse (v_plus [i,j] > 0 && gp [i] > 1, exp (-lambda [i] * ifelse (gp [i] ==  3, phy_dp [i,j], per_dp[i,j,per_choice[i]])), 1)
      s_minus [i,j] <- ifelse (abs(v_minus [i,j]) > 0 && gp [i] > 1, exp (-lambda [i] * ifelse (gp [i] == 3, phy_dm [i,j], per_dm[i,j,per_choice[i]])), 1)

      #------------------------------------------------------------------------------------------------------------#



 #-#-#-#-#-# Learning process #-#-#-#-#-#
      #------------------------------------------------------------------------------------------------------------#
      v_plus [i,j+1] <- ifelse(gp [i] != 1, ifelse(k_plus [i,j] == 1, v_plus [i,j] + alpha[i] * (r_plus [i,j] - v_plus [i,j]),v_plus [i,j]), 0)
	  #-- Excitatory associative strength

      v_minus [i,j+1] <- ifelse(gp [i] != 1, ifelse(k_minus [i,j] == 1, v_minus [i,j] + alpha[i] * (r_minus [i,j] - v_minus [i,j]),v_minus [i,j]), 0)
	  #-- Inhibitory associative strength

	  ## Log-liklihood and prediction for generalization trials
      #------------------------------------------------------------------------------------------------------------#

	  # log-likelihood
      loglik [i,j] <- logdensity.norm(y[i,j],  theta [i,j], 1/sigma [zn [i]]^2)

      # Prediction
      y_pre [i,j] ~ dnorm (theta[i,j] , 1/sigma [zn [i]]^2)


    } # Trial loop end


      v_plus [i,1] <- 0 # The initial excitatory associative strength
      v_minus [i,1] <- 0 # The initial inhibitory associative strength


 #-#-#-#-#-# parameter priors #-#-#-#-#-#
      #------------------------------------------------------------------------------------------------------------#

	  #************************************generalization response****************************************#
      w0 [i] ~ dnorm (w0_mu, 1/w0_sigma^2)
      w1_1 [i] ~ dgamma (w1_a, w1_b) T (1,)
      w1 [i] <- ifelse( gp [i] == 1, 0, w1_1 [i])
      zn [i] <- ifelse (gp [i] == 1, 2, 1)
      choice[i] <- ifelse (gp [i] == 3, 1, 2)
	  #***************************************************************************************************#

	  #************************************generalization process****************************************#
      lambda_1 [i] ~ dnorm (mu_lambda[choice[i]], 1/sigma_lambda[choice[i]] ^2) T(lam_lim[choice[i]],)
      lambda_2 [i] ~ dnorm (mu_lambda[2], 1/sigma_lambda[2] ^2) T (1e-9,lam_lim[choice[i]])
      lambda [i] <- ifelse (gp [i] == 1, 0, ifelse (gp [i] == 2, lambda_2 [i], lambda_1 [i]))
	  #**************************************************************************************************#

	  #************************************learning process****************************************#
      alpha_1 [i] ~ dbeta (a_alpha, b_alpha) T (1e-9,1-1e-9)
      alpha [i] <- ifelse (gp [i] == 1, 0, alpha_1 [i])
	  #********************************************************************************************#


	  #************************************latent group****************************************#
      # gp = 1 - Non-learners : alpha = 0
      # gp = 2 - Overgeneralizers : lambda = 0
      # gp = 3 - Perceptual generalizers : lambda > 0 , d_phy

      gp [i] ~ dcat (pi [])
      per_choice [i] ~ dcat (pi_psi [])
	  #********************************************************************************************#

      #------------------------------------------------------------------------------------------------------------#

    } # Participant loop end

 #-#-#-#-#-# parameter hyperpriors #-#-#-#-#-#
      #------------------------------------------------------------------------------------------------------------#
	  #************************************generalization response****************************************#
      w1_a ~ dgamma (2,1) T (1e-9,)
      w1_b ~ dgamma (2,1) T (1e-9,)
      w0_mu ~ dnorm (0,1/10^2)
      w0_sigma ~ dgamma (2,1)

	  sigma[1] ~ dgamma (2,1)
      sigma[2] ~ dgamma (2,1)

	  lam_lim[1] <- -log(.7)/53.368
	  lam_lim[2] <- -log(.7)
	  #***************************************************************************************************#

	  #************************************generalization process****************************************#
	  mu_lambda[1] ~ dlnorm(-2,pow(.5,-2))
	  mu_lambda[2] ~ dlnorm(1,pow(.5,-2))
      sigma_lambda[1] ~ dunif(1e-9,.5)
      sigma_lambda[2] ~ dunif(1e-9,3)
	  #**************************************************************************************************#

	  #************************************learning process****************************************#
	  a_alpha <- mu_alpha * kappa_alpha
      b_alpha <- (1 - mu_alpha )* kappa_alpha
      mu_alpha ~ dbeta (1,1) T(1e-9,1-1e-9)
      kappa_alpha ~ dunif (1,10)
	  #********************************************************************************************#


	  #************************************latent group********************************************#
      pi[1:4] ~ ddirch(c(1,1,1,1))
	  pi_psi[1:3] ~ ddirch(c(1,1,1))
	  #********************************************************************************************#
    }
